import { afterEach, expect, spyOn, test, vi } from "bun:test";
import { TempDir } from "@oh-my-pi/pi-utils";
import { getBundledModel } from "@oh-my-pi/pi-catalog/models";
import { createStatusCoordinator } from "../../natives/native/status-coordinator.js";
import { ModelRegistry } from "../src/config/model-registry";
import { Settings } from "../src/config/settings";
import { createAgentSession } from "../src/sdk";
import { AuthStorage } from "../src/session/auth-storage";
import { SessionManager } from "../src/session/session-manager";
import { PeerRuntime } from "../src/task/peer-runtime";

afterEach(() => vi.restoreAllMocks());

test("optional peer baseline does not hold concurrent SDK children behind parent status cooldown", async () => {
 using dir = TempDir.createSync("omp-peer-queue-");
 const authStorage = await AuthStorage.create(":memory:");
 const modelRegistry = new ModelRegistry(authStorage, dir.join("models.yml"));
 let now = 0;
 const timers = [];
 const coordinator = createStatusCoordinator({ now: () => now, setTimer: (fn, delay) => { timers.push({ fn, at: now + delay }); return fn; }, clearTimer: () => {} });
 const starts = [];
 await coordinator.request("shared", "normal", "parent", () => { starts.push(now); return ""; });
 let entered = 0;
 spyOn(PeerRuntime.prototype, "initialize").mockImplementation(function () {
  const generation = ++entered;
  return coordinator.request("shared", "all", `child-${generation}`, () => { starts.push(now); return ""; });
 });
 let ready = 0;
 const launches = Array.from({length: 3}, (_, i) => createAgentSession({
  cwd: dir.path(), agentDir: dir.join("agent"), authStorage, modelRegistry,
  settings: Settings.isolated(), sessionManager: SessionManager.inMemory(dir.path()),
  model: getBundledModel("openai", "gpt-4o-mini"), parentTaskPrefix: `peer-${i}`, taskDepth: 1,
  peerCoordination: { task: "fixture", profile: "test", scope: {root: dir.path()}, touched: [] },
  disableExtensionDiscovery: true, skills: [], promptTemplates: [], slashCommands: [], contextFiles: [],
  enableMCP: false, enableLsp: false, skipPythonPreflight: true, toolNames: [],
 }).then(result => { ready++; return result; }));
 try {
  const deadline = performance.now() + 3000;
  while (entered < 3 && performance.now() < deadline) await Bun.sleep(1);
  expect(entered).toBe(3);
  // Parent work outside child launch scopes; optional metadata must stay off readiness.
  const busyUntil = performance.now() + 50;
  while (performance.now() < busyUntil) {}
  await Bun.sleep(20);
  expect(ready).toBe(3);
 } finally {
  for (let i=0; i<6; i++) {
   const timer = timers.shift();
   if (timer) { now = timer.at; timer.fn(); }
   await Bun.sleep(1);
  }
  const results = await Promise.all(launches);
  for (const {session} of results) await session.dispose();
  authStorage.close();
  console.log(JSON.stringify({starts, ready, entered, unit:"simulated ms"}));
 }
 expect(starts).toEqual([0, 10000, 20000, 30000]);
}, 10000);
